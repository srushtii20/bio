# 2. CREATE SALES DATASETS IN CSV, EXCEL AND JSON FORMATS
# ============================================================

    "Order_ID": [101, 102, 103, 104, 105],

    "Unit_Price": [50000, 20000, 1000, 500, 12000],
    "Region": ["Pune", "Mumbai", "Nashik", "Pune", "Mumbai"]
}

excel_data = {
    "Order_ID": [106, 107, 108, 109, 110],
    "Date": ["2026-01-06", "2026-01-07", "2026-01-08",
             "2026-01-09", "2026-01-10"],
    "Product": ["Tablet", "Headphones", "Printer", "Laptop", "Mouse"],
    "Category": ["Electronics", "Accessories", "Electronics",
                 "Electronics", "Accessories"],
    "Quantity": [3, 4, 2, 1, 6],
    "Unit_Price": [25000, 1500, 8000, 50000, 500],
    "Region": ["Nashik", "Pune", "Mumbai", "Pune", "Nashik"]
}

json_data = {
    "Order_ID": [111, 112, 113, 114, 115],
    "Date": ["2026-01-11", "2026-01-12", "2026-01-13",
             "2026-01-14", "2026-01-15"],
    "Product": ["Mobile", "Keyboard", "Monitor", "Tablet", "Printer"],

}

# Convert dictionaries into DataFrames

csv_df = pd.DataFrame(csv_data)
excel_df = pd.DataFrame(excel_data)
json_df = pd.DataFrame(json_data)

# Save datasets in different formats

csv_df.to_csv("sales.csv", index=False)
excel_df.to_excel("sales.xlsx", index=False)
json_df.to_json("sales.json", orient="records", indent=4)

print("CSV, Excel and JSON files created successfully!")

# ============================================================
# 3. LOAD DATA FROM DIFFERENT FILE FORMATS
# ============================================================

csv_df = pd.read_csv("sales.csv")
excel_df = pd.read_excel("sales.xlsx")
json_df = pd.read_json("sales.json")

print("\nCSV DATA:")
print(csv_df.head())

print("\nEXCEL DATA:")
print(excel_df.head())

print("\nJSON DATA:")
print(json_df.head())

# ============================================================
# 4. EXPLORE DATASET STRUCTURE AND CONTENT
# ============================================================

print("\nCSV DATA INFORMATION:")
csv_df.info()

print("\nEXCEL DATA INFORMATION:")
excel_df.info()

print("\nJSON DATA INFORMATION:")
json_df.info()

print("\nMISSING VALUES:")
print(csv_df.isnull().sum())

print("\nSTATISTICAL SUMMARY:")
print(csv_df.describe())

# ============================================================
# 5. COMBINE ALL DATASETS INTO ONE DATAFRAME
# ============================================================

sales_df = pd.concat(
    [csv_df, excel_df, json_df],
    ignore_index=True
)

print("\nCOMBINED SALES DATA:")
print(sales_df)

print("\nTOTAL RECORDS:", len(sales_df))

# ============================================================
# 6. DATA CLEANING AND PREPROCESSING
# ============================================================

# Remove duplicate records

sales_df.drop_duplicates(inplace=True)

# Handle missing values

sales_df["Quantity"] = sales_df["Quantity"].fillna(
    sales_df["Quantity"].median()
)

sales_df["Unit_Price"] = sales_df["Unit_Price"].fillna(
    sales_df["Unit_Price"].median()
)

sales_df["Product"] = sales_df["Product"].fillna("Unknown")
sales_df["Category"] = sales_df["Category"].fillna("Unknown")
sales_df["Region"] = sales_df["Region"].fillna("Unknown")

# Convert date into datetime format

sales_df["Date"] = pd.to_datetime(sales_df["Date"])

# Remove invalid records

sales_df = sales_df[
    (sales_df["Quantity"] > 0) &
    (sales_df["Unit_Price"] >= 0)
].copy()

print("\nCLEANED SALES DATA:")
print(sales_df.head())

print("\nMISSING VALUES AFTER CLEANING:")
print(sales_df.isnull().sum())

# ============================================================
# 7. DATA TRANSFORMATION
# ============================================================

# Calculate total sales

sales_df["Total_Sales"] = (
    sales_df["Quantity"] * sales_df["Unit_Price"]
)

# Extract month and year

sales_df["Month"] = sales_df["Date"].dt.month_name()
sales_df["Year"] = sales_df["Date"].dt.year

# Combine product and category columns

sales_df["Product_Info"] = (
    sales_df["Product"] + " - " + sales_df["Category"]
)

print("\nTRANSFORMED SALES DATA:")
print(sales_df.head())

# ============================================================
# 8. DESCRIPTIVE STATISTICAL ANALYSIS
# ============================================================

total_sales = sales_df["Total_Sales"].sum()

average_order_value = sales_df["Total_Sales"].mean()

maximum_sales = sales_df["Total_Sales"].max()

minimum_sales = sales_df["Total_Sales"].min()

total_quantity = sales_df["Quantity"].sum()

print("\nSALES STATISTICS")

print("Total Sales:", total_sales)

print("Average Order Value:", round(average_order_value, 2))

print("Maximum Sales:", maximum_sales)

print("Minimum Sales:", minimum_sales)

print("Total Quantity Sold:", total_quantity)

print("\nSTATISTICAL SUMMARY:")
print(sales_df.describe())

# ============================================================
# 9. SALES AGGREGATION BY CATEGORY
# ============================================================

category_sales = sales_df.groupby("Category")[
    "Total_Sales"
].sum()

print("\nSALES BY CATEGORY:")
print(category_sales)

# ============================================================
# 10. SALES AGGREGATION BY REGION
# ============================================================

region_sales = sales_df.groupby("Region")[
    "Total_Sales"
].sum()

print("\nSALES BY REGION:")
print(region_sales)

# ============================================================
# 11. SALES AGGREGATION BY PRODUCT
# ============================================================

product_sales = sales_df.groupby("Product")[
    "Total_Sales"
].sum()

print("\nSALES BY PRODUCT:")
print(product_sales)

# ============================================================
# 12. BAR PLOT - SALES BY PRODUCT CATEGORY
# ============================================================

plt.figure(figsize=(8, 5))

category_sales.plot(kind="bar")

plt.title("Sales by Product Category")
plt.xlabel("Category")
plt.ylabel("Total Sales")
plt.xticks(rotation=0)
plt.tight_layout()
plt.show()

# ============================================================
# 13. PIE CHART - PRODUCT CATEGORY DISTRIBUTION
# ============================================================

plt.figure(figsize=(7, 7))

category_sales.plot(
    kind="pie",
    autopct="%1.1f%%",
    startangle=90
)

plt.title("Product Category Distribution")
plt.ylabel("")
plt.show()

# ============================================================
# 14. BAR PLOT - SALES BY REGION
# ============================================================

plt.figure(figsize=(8, 5))

region_sales.plot(kind="bar")

plt.title("Sales by Region")
plt.xlabel("Region")
plt.ylabel("Total Sales")
plt.xticks(rotation=0)
plt.tight_layout()
plt.show()

# ============================================================
# 15. BOX PLOT - SALES DISTRIBUTION
# ============================================================

plt.figure(figsize=(8, 5))

sns.boxplot(x=sales_df["Total_Sales"])

plt.title("Sales Distribution")
plt.xlabel("Total Sales")
plt.show()

# ============================================================
# 16. LINE CHART - SALES TRENDS OVER TIME
# ============================================================

daily_sales = sales_df.groupby("Date")[
    "Total_Sales"
].sum()

plt.figure(figsize=(10, 5))

plt.plot(
    daily_sales.index,
    daily_sales.values,
    marker="o"
)

plt.title("Sales Trends Over Time")
plt.xlabel("Date")
plt.ylabel("Total Sales")
plt.xticks(rotation=45)
plt.grid(True)
plt.tight_layout()
plt.show()

# ============================================================
# 17. SAVE FINAL CLEANED DATASET
# ============================================================

sales_df.to_csv("final_sales.csv", index=False)

sales_df.to_excel("final_sales.xlsx", index=False)

sales_df.to_json(
    "final_sales.json",
    orient="records",
    indent=4,
    date_format="iso"
)

print("\nFinal datasets saved successfully!")
